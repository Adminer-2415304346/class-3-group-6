```powershell
python src/DjangoBlog/manage.py migrate --settings=djangoblog.settings_demo
python doc/week7-8/tools/run_isolated_tests.py servermanager.tests servermanager.test_week78_takeover servermanager.test_week910_functional djangoblog.test_week910_configuration
```
